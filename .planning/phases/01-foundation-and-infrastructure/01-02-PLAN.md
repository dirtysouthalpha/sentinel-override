    **2. db/schema.sql** -- Create DDL with these tables:
    ```sql
    CREATE TABLE IF NOT EXISTS services (
        name TEXT PRIMARY KEY,
        url TEXT NOT NULL,
        status TEXT NOT NULL CHECK(status IN ('unknown', 'healthy', 'degraded', 'unhealthy')),
        last_check_at TIMESTAMP,
        response_time_ms REAL,
        version TEXT,
        critical INTEGER NOT NULL DEFAULT 0,
        poll_interval_seconds INTEGER NOT NULL DEFAULT 30,
        failure_threshold INTEGER NOT NULL DEFAULT 3,
        consecutive_failures INTEGER NOT NULL DEFAULT 0
    );

    CREATE TABLE IF NOT EXISTS health_checks (
        id INTEGER PRIMARY KEY AUTOINCREMENT,
        service_name TEXT NOT NULL,
        status TEXT NOT NULL CHECK(status IN ('healthy', 'degraded', 'unhealthy')),
        response_time_ms REAL,
        checked_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
        details TEXT,
        FOREIGN KEY (service_name) REFERENCES services(name)
    );
    CREATE INDEX IF NOT EXISTS idx_health_service_time ON health_checks(service_name, checked_at);

    CREATE TABLE IF NOT EXISTS recovery_events (
        id INTEGER PRIMARY KEY AUTOINCREMENT,
        service_name TEXT NOT NULL,
        escalation_level INTEGER NOT NULL,
        action TEXT NOT NULL,
        result TEXT NOT NULL,
        snapshot TEXT,
        started_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
        completed_at TIMESTAMP,
        FOREIGN KEY (service_name) REFERENCES services(name)
    );
    CREATE INDEX IF NOT EXISTS idx_recovery_service_time ON recovery_events(service_name, started_at);

    CREATE TABLE IF NOT EXISTS metrics (
        id INTEGER PRIMARY KEY AUTOINCREMENT,
        metric_type TEXT NOT NULL,
        metric_name TEXT NOT NULL,
        value REAL NOT NULL,
        unit TEXT,
        recorded_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
    );
    CREATE INDEX IF NOT EXISTS idx_metrics_type_name_time ON metrics(metric_type, metric_name, recorded_at);

    CREATE TABLE IF NOT EXISTS state_snapshots (
        id INTEGER PRIMARY KEY AUTOINCREMENT,
        snapshot_type TEXT NOT NULL,
        snapshot_data TEXT NOT NULL,
        created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
    );
    CREATE INDEX IF NOT EXISTS idx_snapshots_type_time ON state_snapshots(snapshot_type, created_at);
    ```

    **3. db/connection.py** -- Create:
    - `get_db(db_path: Path)` async context manager:
      - Ensure parent directory exists: `db_path.parent.mkdir(parents=True, exist_ok=True)`
      - `await aiosqlite.connect(db_path)`
      - Execute `PRAGMA journal_mode=WAL;`
      - Execute `PRAGMA synchronous=NORMAL;`
      - Execute `PRAGMA foreign_keys=ON;`
      - Set `db.row_factory = aiosqlite.Row` for dict-like access
      - Yield db connection
      - Close in finally block
    - `init_db(db_path: Path)` async function:

      - Use get_db context manager
      - Execute the full schema SQL with `executescript()`
